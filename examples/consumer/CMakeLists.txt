cmake_minimum_required(VERSION 3.16)
project(consumer C)

find_package(reaktor 0.9 REQUIRED)

add_executable(simple WIN32 ../../samples/simple/simple.c)
option(CONSUMER_SHARED "Link libreaktor.so instead of libreaktor.a" OFF)
if(CONSUMER_SHARED)
    target_link_libraries(simple PRIVATE reaktor::shared)
else()
    target_link_libraries(simple PRIVATE reaktor::reaktor)
endif()
if(EMSCRIPTEN)
    set_target_properties(simple PROPERTIES SUFFIX ".html")
endif()

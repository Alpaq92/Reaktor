function(lcui_expand IN OUTVAR)
    file(READ "${IN}" _c)
    string(REGEX MATCHALL "\\$\\{define [A-Z_]+\\}" _tokens "${_c}")
    foreach(_t IN LISTS _tokens)
        string(REGEX REPLACE "\\$\\{define ([A-Z_]+)\\}" "\\1" _name "${_t}")
        list(FIND LCUI_FEATURES "${_name}" _found)
        if(_found GREATER -1)
            set(_r "#define ${_name} 1")
        else()
            set(_r "/* ${_name}: not enabled by this build */")
        endif()
        string(REPLACE "${_t}" "${_r}" _c "${_c}")
    endforeach()
    string(REPLACE "\${VERSION_MAJOR}" "0" _c "${_c}")
    string(REPLACE "\${VERSION_MINOR}" "0" _c "${_c}")
    string(REPLACE "\${VERSION_ALTER}" "0" _c "${_c}")
    string(REPLACE "\${VERSION}" "vendored" _c "${_c}")
    set(${OUTVAR} "${_c}" PARENT_SCOPE)
endfunction()

add_library(reaktor_locale INTERFACE)
if(NOT REAKTOR_LOCALE)
    target_link_libraries(reaktor_locale INTERFACE reaktor_locale_none)
    message(STATUS "locale: off")
else()
    target_sources(reaktor_locale INTERFACE
        "${_locale}/locale.c" "${_locale}/plural.c" "${_locale}/format.c"
        "${_locale}/locale_runtime.c")
    if(REAKTOR_LOCALE_EMBED)
        file(GLOB _catalogs CONFIGURE_DEPENDS
             "${CMAKE_CURRENT_SOURCE_DIR}/assets/locale/*.txt")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_catalogs})
        set(_gen "/* Generated by CMakeLists.txt from assets/locale. */\n#include \"locale_internal.h\"\n\n")
        set(_rows "")
        set(_n 0)
        foreach(_f IN LISTS _catalogs)
            get_filename_component(_code "${_f}" NAME_WE)
            file(READ "${_f}" _hex HEX)
            file(SIZE "${_f}" _size)
            string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _bytes "${_hex}")
            string(APPEND _gen "static const unsigned char catalog_${_n}[] = {${_bytes}0};\n")
            string(APPEND _rows "    { \"${_code}\", catalog_${_n}, ${_size} },\n")
            math(EXPR _n "${_n} + 1")
        endforeach()
        if(_n EQUAL 0)
            set(_rows "    { 0, 0, 0 },\n")
        endif()
        string(APPEND _gen
            "\nconst struct reaktor_locale_catalog reaktor_locale_catalogs[] = {\n${_rows}};\n"
            "const int reaktor_locale_catalog_count = ${_n};\n")
        set(_out "${CMAKE_BINARY_DIR}/gen/locale_catalogs.c")
        set(_old "")
        if(EXISTS "${_out}")
            file(READ "${_out}" _old)
        endif()
        if(NOT _old STREQUAL _gen)
            file(WRITE "${_out}" "${_gen}")
        endif()
        target_sources(reaktor_locale INTERFACE "${_out}")
        target_compile_definitions(reaktor_locale INTERFACE REAKTOR_LOCALE_EMBEDDED)
        message(STATUS "locale: ${_n} catalogs compiled in")
    else()
        message(STATUS "locale: catalogs read from assets/locale at startup")
    endif()
endif()

function(reaktor_add_app name)
    cmake_parse_arguments(RA "CONSOLE" "" "SOURCES;MODULES;ASSETS" ${ARGN})
    if(RA_CONSOLE)
        add_executable(${name} ${RA_SOURCES} ${REAKTOR_APP_RESOURCES})
    else()
        add_executable(${name} WIN32 ${RA_SOURCES} ${REAKTOR_APP_RESOURCES})
    endif()
    if(NOT RA_MODULES AND NOT "MODULES" IN_LIST RA_KEYWORDS_MISSING_VALUES)
        set(RA_MODULES a11y locale text settings tray popups notify print inspector)
    endif()
    set(_mods "")
    foreach(_m a11y locale text settings tray popups notify print inspector)
        if(_m IN_LIST RA_MODULES)
            list(APPEND _mods reaktor_${_m})
        else()
            list(APPEND _mods reaktor_${_m}_none)
        endif()
    endforeach()
    target_link_libraries(${name} PRIVATE reaktor_runtime ${_mods})
    if(MSVC)
        target_compile_options(${name} PRIVATE /we4013)
    elseif(APPLE)
        target_link_options(${name} PRIVATE -Wl,-dead_strip)
    elseif(NOT EMSCRIPTEN)
        target_link_options(${name} PRIVATE -Wl,--gc-sections)
    endif()
    set_target_properties(${name} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    if(EMSCRIPTEN)
        set(_pre "")
        foreach(_a IN LISTS RA_ASSETS)
            string(REGEX REPLACE "@.*$" "" _src "${_a}")
            string(REGEX REPLACE "^[^@]*" "" _dst "${_a}")
            get_filename_component(_src "${_src}" ABSOLUTE
                                   BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
            list(APPEND _pre "SHELL:--preload-file ${_src}${_dst}")
            if(IS_DIRECTORY "${_src}")
                file(GLOB_RECURSE _deps CONFIGURE_DEPENDS "${_src}/*")
            else()
                set(_deps "${_src}")
            endif()
            set_property(TARGET ${name} APPEND PROPERTY LINK_DEPENDS ${_deps})
        endforeach()
        set_target_properties(${name} PROPERTIES SUFFIX ".html")
        target_compile_options(${name} PRIVATE -Oz)
        target_link_options(${name} PRIVATE
            ${_pre}
            -sALLOW_MEMORY_GROWTH=1
            -sINITIAL_MEMORY=64MB
            -sSTACK_SIZE=4MB
            -sMIN_WEBGL_VERSION=2
            -sMAX_WEBGL_VERSION=2
            -sEXIT_RUNTIME=0
            -Oz
            --closure 1
            "SHELL:--shell-file ${REAKTOR_SOURCE_DIR}/tools/shell.html")
        set_property(TARGET ${name} APPEND PROPERTY LINK_DEPENDS
                     "${REAKTOR_SOURCE_DIR}/tools/shell.html"
                     "${REAKTOR_SOURCE_DIR}/tools/web-pre.js")
    endif()
endfunction()

function(reaktor_outside_links)
    set(_libs "")
    set(_dirs "")
    set(_opts "")
    set(_weak "")
    set(_seen "")
    set(_refound "")
    set(_todo ${ARGN})
    while(_todo)
        list(POP_FRONT _todo _t)
        get_target_property(_real ${_t} ALIASED_TARGET)
        if(_real)
            set(_t ${_real})
        endif()
        if(_t IN_LIST _seen)
            continue()
        endif()
        list(APPEND _seen ${_t})
        get_target_property(_type ${_t} TYPE)
        get_target_property(_imported ${_t} IMPORTED)
        set(_items "")
        if(_imported AND NOT _type STREQUAL "INTERFACE_LIBRARY")
            set(_props IMPORTED_IMPLIB_RELEASE IMPORTED_IMPLIB IMPORTED_LOCATION_RELEASE
                       IMPORTED_LOCATION)
            get_target_property(_cfgs ${_t} IMPORTED_CONFIGURATIONS)
            if(_cfgs)
                foreach(_c IN LISTS _cfgs)
                    list(APPEND _props IMPORTED_IMPLIB_${_c} IMPORTED_LOCATION_${_c})
                endforeach()
            endif()
            set(_file "")
            foreach(_p IN LISTS _props)
                get_target_property(_f ${_t} ${_p})
                if(_f)
                    set(_file "${_f}")
                    break()
                endif()
            endforeach()
            if(_file)
                list(APPEND _libs "${_file}")
            else()
                message(WARNING "libreaktor finds no file for ${_t}, which its parts link")
            endif()
        elseif(_t IN_LIST ARGN)
            get_target_property(_v ${_t} LINK_LIBRARIES)
            if(_v)
                list(APPEND _items ${_v})
            endif()
        elseif(NOT _type MATCHES "^(INTERFACE|OBJECT)_LIBRARY$")
            message(WARNING "libreaktor leaves out ${_t}, which its parts link")
        endif()
        foreach(_p INTERFACE_LINK_LIBRARIES INTERFACE_LINK_OPTIONS INTERFACE_LINK_DIRECTORIES)
            get_target_property(_v ${_t} ${_p})
            if(NOT _v)
                continue()
            endif()
            if(_p STREQUAL "INTERFACE_LINK_LIBRARIES")
                list(APPEND _items ${_v})
            elseif(_v MATCHES "\\$<")
                message(WARNING "libreaktor cannot carry ${_t}'s ${_p}: ${_v}")
            elseif(_p STREQUAL "INTERFACE_LINK_OPTIONS")
                list(APPEND _opts ${_v})
            else()
                list(APPEND _dirs ${_v})
            endif()
        endforeach()
        foreach(_i IN LISTS _items)
            if(_i MATCHES "^\\$<LINK_ONLY:(.*)>$")
                set(_i "${CMAKE_MATCH_1}")
            endif()
            if(_i MATCHES "^\\$<(TARGET_NAME|BUILD_INTERFACE):(.*)>$")
                set(_i "${CMAKE_MATCH_2}")
            endif()
            if(_i MATCHES "^\\$<INSTALL_INTERFACE:")
                continue()
            endif()
            # A dependency's own directory may have found it: its targets are not seen here.
            if(NOT TARGET "${_i}" AND _i MATCHES "^([A-Za-z0-9_]+)::")
                set(_pkg "${CMAKE_MATCH_1}")
                if(NOT _pkg STREQUAL "PkgConfig" AND NOT _pkg IN_LIST _refound)
                    list(APPEND _refound ${_pkg})
                    find_package(${_pkg} QUIET)
                endif()
            endif()
            if(TARGET "${_i}")
                list(APPEND _todo "${_i}")
            elseif(_i MATCHES "^PkgConfig::(.+)$")
                set(_pc_prefix "${CMAKE_MATCH_1}")
                list(APPEND _libs ${${_pc_prefix}_LIBRARIES})
                list(APPEND _dirs ${${_pc_prefix}_LIBRARY_DIRS})
                list(APPEND _opts ${${_pc_prefix}_LDFLAGS_OTHER})
            elseif(_i MATCHES "::")
                message(WARNING "libreaktor cannot find ${_i}, which ${_t} links")
            elseif(_i MATCHES "^\\$<LINK_LIBRARY:FRAMEWORK,(.*)>$")
                list(APPEND _libs "-framework ${CMAKE_MATCH_1}")
            elseif(_i MATCHES "^\\$<LINK_LIBRARY:WEAK_FRAMEWORK,(.*)>$")
                list(APPEND _weak "${CMAKE_MATCH_1}")
            elseif(_i MATCHES "\\$<")
                message(WARNING "libreaktor cannot carry ${_t}'s link to ${_i}")
            elseif(_i)
                list(APPEND _libs "${_i}")
            endif()
        endforeach()
    endwhile()

    # A shared library found on this machine is named, so the package finds it on another.
    set(_named "")
    foreach(_l IN LISTS _libs)
        if(IS_ABSOLUTE "${_l}" AND _l MATCHES "/lib([^/]+)\\.(so|dylib|tbd)(\\.[0-9.]+)?$")
            list(APPEND _named "${CMAKE_MATCH_1}")
            get_filename_component(_d "${_l}" DIRECTORY)
            string(FIND "${_d}" "${CMAKE_OSX_SYSROOT}" _in_sdk)
            if(NOT (CMAKE_OSX_SYSROOT AND _in_sdk EQUAL 0))
                list(APPEND _dirs "${_d}")
            endif()
        else()
            list(APPEND _named "${_l}")
        endif()
    endforeach()
    list(REMOVE_DUPLICATES _named)
    # A cross build found them under the sysroot; the package names where they are on the target.
    if(CMAKE_SYSROOT)
        set(_stripped "")
        foreach(_d IN LISTS _dirs)
            string(FIND "${_d}" "${CMAKE_SYSROOT}" _in_sysroot)
            if(_in_sysroot EQUAL 0)
                string(LENGTH "${CMAKE_SYSROOT}" _n)
                string(SUBSTRING "${_d}" ${_n} -1 _d)
            endif()
            list(APPEND _stripped "${_d}")
        endforeach()
        set(_dirs ${_stripped})
    endif()
    list(REMOVE_DUPLICATES _dirs)
    foreach(_d IN LISTS CMAKE_C_IMPLICIT_LINK_DIRECTORIES CMAKE_PLATFORM_IMPLICIT_LINK_DIRECTORIES)
        list(REMOVE_ITEM _dirs "${_d}")
    endforeach()
    list(REMOVE_DUPLICATES _opts)
    list(REMOVE_DUPLICATES _weak)

    set(_pc "")
    foreach(_d IN LISTS _dirs)
        string(APPEND _pc " -L${_d} -Wl,-rpath,${_d}")
    endforeach()
    foreach(_l IN LISTS _named)
        if(_l MATCHES "^-" OR IS_ABSOLUTE "${_l}")
            string(APPEND _pc " ${_l}")
        else()
            string(APPEND _pc " -l${_l}")
        endif()
    endforeach()
    foreach(_w IN LISTS _weak)
        list(APPEND _opts "SHELL:-Xlinker -weak_framework -Xlinker ${_w}")
    endforeach()
    foreach(_o IN LISTS _opts)
        string(REGEX REPLACE "^SHELL:" "" _o "${_o}")
        string(REGEX REPLACE "-Xlinker -weak_framework -Xlinker " "-Wl,-weak_framework," _o "${_o}")
        string(APPEND _pc " ${_o}")
    endforeach()

    set(REAKTOR_LINK_LIBS "${_named}" PARENT_SCOPE)
    set(REAKTOR_LINK_DIRS "${_dirs}" PARENT_SCOPE)
    set(REAKTOR_OUTSIDE_OPTIONS "${_opts}" PARENT_SCOPE)
    set(REAKTOR_PC_LIBS "${_pc}" PARENT_SCOPE)
endfunction()

    COMMAND "${CMAKE_COMMAND}" -E remove -f "${_lib}"
    COMMAND ${_merge}
    DEPENDS ${_parts}
    COMMENT "Merging libreaktor"
    VERBATIM)
add_custom_target(libreaktor DEPENDS "${_lib}")

option(REAKTOR_INSTALL "Install libreaktor's files" ${_reaktor_top})
if(REAKTOR_INSTALL)
    include(CMakePackageConfigHelpers)
    configure_file(tools/reaktorConfig.cmake.in "${CMAKE_BINARY_DIR}/reaktorConfig.cmake" @ONLY)
    write_basic_package_version_file("${CMAKE_BINARY_DIR}/reaktorConfigVersion.cmake"
        COMPATIBILITY SameMinorVersion)
    install(FILES "${_lib}" DESTINATION lib COMPONENT libreaktor OPTIONAL)
    install(DIRECTORY include/reaktor DESTINATION include COMPONENT libreaktor)
    install(FILES "${TP}/nuklear/nuklear.h" DESTINATION include COMPONENT libreaktor)
    install(FILES "${TP}/nukanim/nukanim.h" DESTINATION include COMPONENT libreaktor)
    install(DIRECTORY "${TP}/SDL/include/SDL3" DESTINATION include COMPONENT libreaktor)
    install(FILES "${CMAKE_BINARY_DIR}/reaktorConfig.cmake"
                  "${CMAKE_BINARY_DIR}/reaktorConfigVersion.cmake"
            DESTINATION lib/cmake/reaktor COMPONENT libreaktor)
    if(NOT MSVC)
        configure_file(tools/reaktor.pc.in "${CMAKE_BINARY_DIR}/reaktor.pc" @ONLY)
        install(FILES "${CMAKE_BINARY_DIR}/reaktor.pc" DESTINATION lib/pkgconfig
                COMPONENT libreaktor)
    endif()
    install(DIRECTORY "${CMAKE_BINARY_DIR}/licenses/" DESTINATION share/reaktor/licenses
            COMPONENT libreaktor)
    install(FILES assets/fonts/Aileron-Notice.txt DESTINATION share/reaktor/licenses
            COMPONENT libreaktor)
    install(FILES docs/NOTICE.md DESTINATION share/reaktor COMPONENT libreaktor)
    if(EMSCRIPTEN)
        install(FILES tools/web-pre.js DESTINATION share/reaktor COMPONENT libreaktor)
    endif()
    if(UNIX AND NOT APPLE AND NOT EMSCRIPTEN)
        foreach(_fmt deb rpm)
            add_custom_target(${_fmt}
                COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/tools/libreaktor-pkg.sh"
                        -f ${_fmt} -b "${CMAKE_BINARY_DIR}" -o "${CMAKE_BINARY_DIR}/dist"
                DEPENDS libreaktor
                COMMENT "Packing libreaktor as a .${_fmt}"
                USES_TERMINAL VERBATIM)
        endforeach()
    endif()
endif()
